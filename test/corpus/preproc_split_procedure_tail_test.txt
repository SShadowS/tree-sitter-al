; A split procedure signature is followed by exactly the text that follows an
; ordinary one, so preproc_split_procedure shares procedure's _procedure_tail.
; Before, it accepted only `[;] [var] begin ... end`: cases 1 and 2 were ERROR
; trees, and case 3 parsed the pragma-only #if as an empty
; preproc_conditional_var_block. alc accepts all three, four-way (flat and split,
; FOO defined and undefined).

========================================================================
Split signature followed by a split procedure body
========================================================================
codeunit 50100 Probe
{
#if FOO
    procedure Foo(a: Integer)
#else
    procedure Foo(a: Integer; b: Integer)
#endif
#if FOO
    var
        i: Integer;
    begin
        i := a;
#else
    begin
#endif
        Message('x');
    end;
}
