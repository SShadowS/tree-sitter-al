========================================================================
Preprocessor with OR condition
========================================================================
codeunit 1 T
{
#if BCONLINE or ONPREM
    var x: Integer;
#endif
}
------------------------------------------------------------------------

(source_file
  (codeunit_declaration
    (codeunit_keyword)
    object_id: (integer)
    object_name: (identifier)
    body: (declaration_body
      (preproc_conditional
        (preproc_if
          (preproc_open)
          condition: (preproc_or_expression
            (identifier)
            (identifier)))
        (var_section
          (var_keyword)
          body: (var_body
            (variable_declaration
              name: (identifier)
              type: (type_specification
                (basic_type)))))
        (preproc_endif
          (preproc_close))))))
