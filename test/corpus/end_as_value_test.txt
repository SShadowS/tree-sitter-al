================================================================================
End as a value keeps parsing after the #40/#41 end check
--
The #41 scanner check kills `end` read as a name where a statement can start
(BEGIN_KEYWORD valid, END_KEYWORD not). alc builds `End` as a VALUE, and no
large corpus holds one, so this fixture is the gate: OptionMembers, InitValue,
ValuesAllowed, TableRelation const(), CalcFormula filter(), SourceTableView
const() and RunPageLink const() each read `End` as an identifier. alc accepts
it (tools/alc_probe/cases/issue40-42/end_as_value.al); the tree equals 4.5.0's.
alc rejects `End` as a variable name (AL0519), so the check takes nothing there.
================================================================================
table 50000 EndValues
{
    fields
    {
        field(1; Kind; Option)
        {
            OptionMembers = End,Start;
            InitValue = End;
        }
        field(2; Other; Option)
        {
            OptionMembers = End,Start;
            ValuesAllowed = End;
        }
        field(3; Code; Code[20])
        {
            TableRelation = EndValues.Code where(Kind = const(End));
        }
        field(4; Cnt; Integer)
        {
            FieldClass = FlowField;
            CalcFormula = count(EndValues where(Kind = filter(End)));
        }
    }
}

page 50000 EndPage
{
    SourceTable = EndValues;
    SourceTableView = where(Kind = const(End));

    actions
    {
        area(Processing)
        {
            action(Open)
            {
                RunObject = page EndPage;
                RunPageLink = Kind = const(End);
            }
        }
    }
}
