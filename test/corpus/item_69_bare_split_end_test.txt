================================================================================
Item 69: a bare `end` in both arms of a split block, the `;` after #endif
--
Both `end`s used to ERROR: PREPROC_SPLIT_END wanted `end ;` before the
directive, and #41 had killed the call_statement misreading that hid it. The
`;` is now optional in the scanner's lookahead; the `;` after #endif is the
host's separator. alc accepts X=0 and X=1
(tools/alc_probe/cases/item69/bare-end-else.al).
================================================================================
codeunit 1 C
{
    procedure P(R: Boolean)
    begin
        if R then begin
            A();
#if X
        end
#else
        end
#endif
        ;
        B();
    end;
}
