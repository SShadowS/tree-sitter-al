Branches ending in an open statement prefix, completed after #endif
================================================================================
codeunit 50100 T
{
    procedure MixedBranches(A: Integer; K: Integer)
    var
        C: Integer;
    begin
        case K of
            1:
#if not C28
                if A = 0 then begin
                    C := 1;
                end else
#else
                if A <> 0 then
#endif
                    C := 2;
            2:
                C := 3;
        end;
    end;

    procedure PreambleElseBegin(A: Integer; B: Boolean)
    var
        C: Integer;
    begin
        repeat
#if not C28
            C := 0;
            if B then
                Message('m')
            else begin
#endif
                C := 1;
                if C > 0 then
                    C := 2;
#if not C28
            end;
#endif
        until A = 0;
    end;

    procedure BeginThenOpenElse(Handled: Boolean; Y: Boolean)
    var
        C: Integer;
    begin
        if not Handled then
#if not C28
        begin
            C := 1;
            if Y then begin
                C := 2;
            end else
#endif
                C := 3;
#if not C28
        end;
#endif
        C := 4;
    end;

    procedure ElseAlternation(N: Integer)
    var
        C: Integer;
    begin
        if N < 1 then
            Message('a')
#if not C28
        else begin
#else
        else
#endif
            for C := 1 to 2 do begin
                Message('b');
            end;
#if not C28
            Message('c');
        end;
#endif
    end;

    procedure CaseElseAlternation(K: Integer)
    var
        C: Integer;
        H: Boolean;
    begin
        case K of
            1:
                C := 1
#if C28
            else
#else
            else begin
                H := false;
#endif
#if not C28
                if not H then
#endif
                    C := 2;
#if not C28
            end;
#endif
        end;
    end;
}
