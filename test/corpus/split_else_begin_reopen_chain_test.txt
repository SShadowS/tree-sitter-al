    var
        C: Integer;
    begin
        if A then begin
            C := 1;
            if B then begin
                C := 2;
#if not C28

            end;
        end else
            if L then begin
                C := 4;
                if S then begin
                    C := 5;
#endif
                end;
            end;
        C := 6;
    end;
}
